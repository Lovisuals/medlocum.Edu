SECTION D — DATA MODEL EXTENSIONS

D.1 Additional Tables (append to LMS_SKILL_UNIVERSAL.md Section 5 schema)

```sql
CREATE TABLE placement_readiness (
  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id               UUID REFERENCES users(id) ON DELETE CASCADE,
  is_ready              BOOLEAN DEFAULT FALSE,
  mandatory_complete    INT DEFAULT 0,
  role_modules_complete INT DEFAULT 0,
  blockers              JSONB DEFAULT '[]',
  assessed_at           TIMESTAMPTZ DEFAULT NOW(),
  tenant_id             UUID NOT NULL
);

CREATE TABLE cpd_summary (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id           UUID REFERENCES users(id) ON DELETE CASCADE,
  year              INT NOT NULL,
  hours_courses     NUMERIC(6,2) DEFAULT 0,
  hours_manual      NUMERIC(6,2) DEFAULT 0,
  hours_total       NUMERIC(6,2) DEFAULT 0,
  nmc_requirement   NUMERIC(6,2) DEFAULT 35,
  gmc_requirement   NUMERIC(6,2) DEFAULT 50,
  hcpc_requirement  NUMERIC(6,2) DEFAULT 0,
  updated_at        TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(user_id, year)
);

CREATE TABLE placement_history (
  id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id        UUID REFERENCES users(id) ON DELETE CASCADE,
  organisation   VARCHAR(300),
  specialty      VARCHAR(150),
  grade          VARCHAR(100),
  start_date     DATE NOT NULL,
  end_date       DATE,
  compliance_snapshot JSONB,
  tenant_id      UUID NOT NULL
);
```

D.2 Additional API Endpoints (append to Section 6.2)

```
PLACEMENT READINESS
  GET  /placement/readiness/:userId  → PlacementReadinessReport
  POST /placement/readiness/check    → PlacementReadinessReport

CPD SUMMARY
  GET  /cpd/summary?year=            → CpdSummary
  GET  /cpd/summary/pdf              → Binary PDF

ADMIN — COMPLIANCE MATRIX
  GET  /admin/compliance/matrix      → ComplianceMatrix[]
  GET  /admin/compliance/at-risk     → User[] (overdue > 7 days)
```
